export type AssetClass = 'index' | 'crypto' | 'commodity' | 'forex';

export interface MarketQuote {
  symbol: string;          // "SPX"
  name: string;            // "S&P 500"
  price: number;
  change: number;
  changePct: number;
  assetClass: AssetClass;
  spark: number[];         // 30-pt intraday series
  updatedAt: string;       // ISO 8601
}

export type Topic = 'finance' | 'tech' | 'auto' | 'articles';

export interface NewsItem {
  id: string;
  topic: Topic;
  subTopic: string;        // "AI" | "EV" | "Monetary Policy"
  title: string;
  summary?: string;
  source: string;         // "Reuters"
  url: string;
  publishedAt: string;
  isBreaking?: boolean;
  severity?: 'info' | 'warn' | 'critical';
}

export type ExperienceLevel = 'Intern' | 'Entry' | 'Mid' | 'Senior' | 'Lead';
export type Domain = 'Engineering' | 'Quant' | 'Finance'
             | 'Data' | 'Product' | 'Design' | 'Risk' | 'Research';

export interface JobPosting {
  id: string;
  title: string;
  firm: string;              // "J.P. Morgan"
  firmLogo: string;          // CDN url or monogram
  location: string;          // "London, UK"
  isRemote: boolean;
  division: string;          // "Global Markets"
  experience: ExperienceLevel;
  domain: Domain;
  postedAt: string;
  applyUrl: string;
}

export interface FeedEvent {
  id: string;
  kind: 'breaking' | 'market' | 'tech' | 'auto' | 'policy';
  message: string;
  at: string;

// app/api/markets/route.ts — cached aggregator with mock fallback
import { NextResponse } from 'next/server';
import { getQuotes } from '@/lib/services/finnhub';
import { MOCK_QUOTES } from '@/lib/mock/quotes';

// 60-second TTL for live tickers
export const revalidate = 60;
export const runtime = 'edge';

const SYMBOLS = ['SPX', 'NDX', 'NIFTY', 'UKX', 'BTC', 'ETH'];

export async function GET() {
  try {
    const quotes = await getQuotes(SYMBOLS);

    return NextResponse.json(
      { ok: true, data: quotes, ts: Date.now() },
      { headers: {
          'Cache-Control': 's-maxage=60, stale-while-revalidate=120'
      }}
    );
  } catch (err) {
    console.error('[markets] upstream failed, serving mock', err);

    // Graceful degradation — never a blank dashboard
    return NextResponse.json({ ok: false, data: MOCK_QUOTES, ts: Date.now() });
  }
}

// app/api/stream/route.ts — Server-Sent Events live ticker
export const dynamic = 'force-dynamic';

export async function GET(req: Request) {
  const encoder = new TextEncoder();

  const stream = new ReadableStream({
    async start(controller) {
      const send = (payload: unknown) =>
        controller.enqueue(
          encoder.encode(`data: ${JSON.stringify(payload)}\n\n`)
        );

      const tick = setInterval(async () => {
        const quotes = await getQuotes(SYMBOLS);
        send({ type: 'quotes', data: quotes });
      }, 2000);

      req.signal.addEventListener('abort', () => {
        clearInterval(tick);
        controller.close();
      });
    },
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache, no-transform',
      'Connection': 'keep-alive',
    },
  });
}

// hooks/useLiveTicker.ts — client side
export function useLiveTicker(initial: MarketQuote[]) {
  const [quotes, setQuotes] = useState(initial);

  useEffect(() => {
    const es = new EventSource('/api/stream');
    es.onmessage = (e) => {
      const msg = JSON.parse(e.data);
      if (msg.type === 'quotes') setQuotes(msg.data);
    };
    return () => es.close();
  }, []);

  return quotes;
}
